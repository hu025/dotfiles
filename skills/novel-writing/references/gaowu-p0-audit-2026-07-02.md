## 批量验证命令

```python
# 1. 词条系统最后出现章节
last_sys = 0
for ch in range(1, 66):
    matching = [f for f in os.listdir(base) if f.startswith(f'第{ch}章')]
    if matching:
        with open(os.path.join(base, matching[0])) as fh:
            if '词条' in fh.read():
                last_sys = ch
print(f"词条系统最后出现: 第{last_sys}章")

# 2. 角色章节分布+断档检测
roles = {'林默': [], '冰璃': [], '陈霖': [], '林天豪': [], '苏清月': [], '苏映雪': [], '韩奎': [], '陈傲': []}
for f in files:
    ch = int(re.search(r'第(\d+)章', f).group(1))
    with open(os.path.join(base, f)) as fh:
        c = fh.read()
    for role in roles:
        if role in c:
            roles[role].append(ch)
for role, chapters in roles.items():
    gaps = [(chapters[i], chapters[i+1]) for i in range(len(chapters)-1) if chapters[i+1]-chapters[i] > 5]
    if gaps:
        print(f"{role}断档: {gaps}")

# 3. em-dash热点
for f in files:
    with open(os.path.join(base, f)) as fh:
        c = fh.read()
    em = c.count('\u2014') + c.count('\u2013')
    if em > 50:
        ch = re.search(r'第(\d+)章', f).group(1)
        print(f"第{ch}章: {em}处em-dash")
```

1. **书名核心卖点必须贯穿至结局** — "词条顿悟系统"是书名来源，不能在最高潮章节消失
2. **威胁≠挖坑，挖坑必须填** — 第59章的"所有人都会复活"是重大悬念，60-65章一个字没提是P0级
3. **重要NPC须有独立场景** — 母亲苏映雪全书仅2处，情感线完全断裂
